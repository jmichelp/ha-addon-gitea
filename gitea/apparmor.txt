#include <tunables/global>

profile gitea flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>
  #include <abstractions/nameservice>

  # Capabilities
  file,
  signal (send, receive),
  capability dac_override,
  capability dac_read_search,
  capability fowner,
  capability fsetid,
  capability chown,
  capability setgid,
  capability setuid,
  capability kill,
  capability net_bind_service,

  # S6-Overlay and binaries
  /init ix,
  /bin/** ix,
  /usr/bin/** ix,
  /run/{s6,s6-rc*,service}/** ix,
  /package/** ix,
  /command/** ix,
  /etc/s6-overlay/** ix,
  /etc/services.d/** rwix,
  /etc/cont-init.d/** rwix,
  /etc/cont-finish.d/** rwix,
  /run/{,**} rwk,
  /dev/tty rw,
  /dev/null rw,
  /dev/urandom r,

  # Bashio
  /usr/lib/bashio/** ix,
  /tmp/** rwk,

  # Mapped volumes and configuration
  /data/** rwkl,
  /ssl/** r,
  /share/** rwkl,

  # Start new profile for service
  /usr/bin/gitea cx -> gitea_bin,
  /usr/local/bin/gitea cx -> gitea_bin,
  /app/gitea/gitea cx -> gitea_bin,

  profile gitea_bin flags=(attach_disconnected,mediate_deleted) {
    #include <abstractions/base>
    #include <abstractions/nameservice>
    #include <abstractions/user-tmp>

    # Signals
    signal (receive),
    signal (send),

    # Networking
    network tcp,
    network udp,
    network inet,
    network inet6,

    # Persistent data and mapped volumes
    /data/** rwkl,
    /share/** rwkl,
    /ssl/** r,

    # Access required for service functionality
    /usr/bin/gitea rix,
    /usr/local/bin/gitea rix,
    /app/gitea/gitea rix,
    /bin/bash rix,
    /bin/sh rix,
    /bin/echo ix,
    /etc/passwd r,
    /etc/group r,
    /dev/tty rw,
    /dev/null rw,
    /dev/urandom r,
    /dev/random r,
    /etc/mime.types r,
    /etc/ssl/** r,
    /etc/ca-certificates/** r,
    /etc/gitea/** r,
    /var/lib/gitea/** rwkl,
    /proc/** r,
    /sys/** r,
    /usr/bin/** rix,
    /usr/libexec/git-core/** rix,
    /usr/lib/git-core/** rix,
    /usr/share/** r,

    /tmp/** rwkl,

    owner @{HOME}/** rwkl,
    owner @{HOME}/ rwkl,
    owner /data/** rwkl,
    owner /data/ rwkl,
  }
}
